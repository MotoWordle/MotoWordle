    function updateHistoryDisplay() {

      const list = document.getElementById('historyList');

      if (guessHistory.length === 0) {

        list.innerHTML = '<div class="guess-item">No guesses yet</div>';

        return;

      }


      list.innerHTML = guessHistory.map((g, i) => {

        const makeIcon = g.makeCorrect ? '<span class="correct">✓</span>' : '<span class="wrong">✗</span>';

        const modelIcon = g.modelCorrect ? '<span class="correct">✓</span>' : '<span class="wrong">✗</span>';

        const yearIcon = g.yearCorrect ? '<span class="correct">✓</span>' : '<span class="wrong">✗</span>';

        return `<div class="guess-item">

          <strong>${i + 1}.</strong> 

          ${g.make} ${makeIcon} &nbsp; 

          ${g.model} ${modelIcon} &nbsp; 

          ${g.year || '—'} ${yearIcon}

        </div>`;

      }).join('');

    }


    function calculateYearPoints(guessYear, actualYear) {

      if (isNaN(guessYear) || isNaN(actualYear)) return 0;

      const diff = Math.abs(guessYear - actualYear);

      if (diff > 20) return 0;

      return Math.max(0, 20 - diff);

    }


    function submitGuess() {

      if (tries <= 0 || solved) return;


      const guessMake = makeSelect.value;

      const guessModel = modelSelect.value;

      const guessYear = parseInt(document.getElementById('year').value);


      if (!guessMake || !guessModel) {

        document.getElementById('message').textContent = "Please select both Make and Model.";

        document.getElementById('message').className = "fail";

        return;

      }


      const makeCorrect = guessMake === dailyBike.make;

      const modelCorrect = guessModel === dailyBike.model;

      const yearCorrect = guessYear === dailyBike.year;

      const yearPoints = calculateYearPoints(guessYear, dailyBike.year);


      // Feedback icons under each field

      makeFeedback.textContent = makeCorrect ? '✓' : '✗';

      makeFeedback.className = 'feedback-icon ' + (makeCorrect ? 'correct' : 'wrong');


      modelFeedback.textContent = modelCorrect ? '✓' : '✗';

      modelFeedback.className = 'feedback-icon ' + (modelCorrect ? 'correct' : 'wrong');


      yearFeedback.textContent = yearCorrect ? '✓' : '✗';

      yearFeedback.className = 'feedback-icon ' + (yearCorrect ? 'correct' : 'wrong');


      // Save to history

      guessHistory.push({

        make: guessMake,

        model: guessModel,

        year: guessYear || null,

        makeCorrect,

        modelCorrect,

        yearCorrect

      });

      updateHistoryDisplay();


      const messageEl = document.getElementById('message');

      const scoreEl = document.getElementById('scoreDisplay');

      guessesUsed = 6 - tries;


      if (makeCorrect && modelCorrect) {

        finalScore = 80 + yearPoints;

        solved = true;

        document.getElementById('submitBtn').disabled = true;

        

        img.style.transformOrigin = "center center";

        img.style.transform = "scale(1)";

        

        document.getElementById('shareBtn').style.display = "inline-block";


        let yearMsg = yearPoints === 20 ? " Perfect year!" :

                      yearPoints > 0   ? ` Year was ${20 - yearPoints} year(s) off.` :

                                         " Year >20 years off (0 pts).";


        messageEl.textContent = `Correct! ${dailyBike.year} ${dailyBike.make} ${dailyBike.model}.${yearMsg}`;

        messageEl.className = "success";

        scoreEl.textContent = `Score: ${finalScore} / 100`;


        streak++;

        localStorage.setItem('motowordleStreak', streak);

        document.getElementById('streak').textContent = streak;

      } else {

        tries--;

        document.getElementById('tries').textContent = tries;

        updateZoom();

        

        messageEl.textContent = "Incorrect. Zooming out…";

        messageEl.className = "fail";

        scoreEl.textContent = "Score: 0 / 100";


        if (tries === 0) {

          img.style.transformOrigin = "center center";

          img.style.transform = "scale(1)";

          messageEl.textContent = `Out of tries. It was a ${dailyBike.year} ${dailyBike.make} ${dailyBike.model}.`;

          document.getElementById('submitBtn').disabled = true;

          document.getElementById('shareBtn').style.display = "inline-block";

          streak = 0;

          localStorage.setItem('motowordleStreak', 0);

          document.getElementById('streak').textContent = 0;

          finalScore = 0;

          guessesUsed = 5;

        }

      }

    }


    function revealAnswer() {

      if (solved) return;

      img.style.transformOrigin = "center center";

      img.style.transform = "scale(1)";

      document.getElementById('message').textContent = `Answer: ${dailyBike.year} ${dailyBike.make} ${dailyBike.model}`;

      document.getElementById('message').className = "fail";

      document.getElementById('submitBtn').disabled = true;

      document.getElementById('tries').textContent = 0;

      document.getElementById('shareBtn').style.display = "inline-block";

      streak = 0;

      localStorage.setItem('motowordleStreak', 0);

      document.getElementById('streak').textContent = 0;

      finalScore = 0;

      guessesUsed = 5;

    }


    function shareResults() {

      const today = new Date().toISOString().slice(0, 10);

      const emojiLine = solved 

        ? "🟩".repeat(guessesUsed) + "⬜".repeat(5 - guessesUsed)

        : "🟥".repeat(5);


      const text = `🏍️ MotoWordle ${today}

Score: ${finalScore}/100

${emojiLine}

Tries: ${guessesUsed}/5

🔥 Streak: ${streak}`;


      navigator.clipboard.writeText(text).then(() => {

        document.getElementById('message').textContent = "Results copied to clipboard!";

        document.getElementById('message').className = "success";
